# OkapiaBench: a self-running benchmark for a 68k or PowerPC Macintosh, built with Retro68.
# Copyright (C) 2026  Jonathan Pepin
# SPDX-License-Identifier: MIT
# scripts/build-okapiabench.sh runs this once per toolchain and joins the two
# into one fat application; by hand:
#   cmake -S . -B build68k -DCMAKE_TOOLCHAIN_FILE=<Retro68-build>/toolchain/m68k-apple-macos/cmake/retro68.toolchain.cmake
#   cmake -S . -B buildppc -DCMAKE_TOOLCHAIN_FILE=<Retro68-build>/toolchain/powerpc-apple-macos/cmake/retroppc.toolchain.cmake
cmake_minimum_required(VERSION 3.19)
project(OkapiaBench C)

set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(EMBENCH ${ROOT}/external/embench-iot)
set(COREMARK ${ROOT}/external/coremark)
set(MINIZ ${ROOT}/external/miniz)
set(STB ${ROOT}/external/stb)
option(NO_EMBENCH "build without Embench (GPL-3.0-or-later)" OFF)
set(REFERENCE "" CACHE FILEPATH "a header from run.py --emit-header: the recap's scores against it")

if(CMAKE_SYSTEM_NAME MATCHES RetroPPC)
    set(PPC ON)
    # No multiply-add fused behind our back: the same double results as a
    # 68k with an exact FPU would round to, where the 68881 does not intervene.
    set(INT_FLAGS -ffp-contract=off)
    set(SOFT_FLAGS -ffp-contract=off)
    set(FPU_FLAGS -ffp-contract=off)
else()
    set(PPC OFF)
    # The tests as 68020 code, as compiled applications of the 1990s were;
    # floating point only where an FPU is required (fp.c, fpu68881.c,
    # imgenc_fpu.c), soft everywhere else so that a 68020 or 68030 without
    # one runs the rest. main() stays 68000 code to refuse a 68000 politely.
    set(INT_FLAGS -m68020 -msoft-float)
    set(SOFT_FLAGS -m68020 -msoft-float)
    set(FPU_FLAGS -m68020 -m68881)
endif()

# Embench's reference times, for the recap's score
file(READ ${EMBENCH}/baseline-data/speed.json SPEED_JSON)
string(JSON SPEED_COUNT LENGTH "${SPEED_JSON}")
math(EXPR SPEED_LAST "${SPEED_COUNT} - 1")
set(EMB_BASELINE "/* generated by CMakeLists.txt from external/embench-iot/baseline-data/speed.json */\n")
string(APPEND EMB_BASELINE "static const struct { const char *name; unsigned long ms; } embBaseline[] = {\n")
foreach(i RANGE ${SPEED_LAST})
    string(JSON key MEMBER "${SPEED_JSON}" ${i})
    string(JSON ms GET "${SPEED_JSON}" "${key}")
    string(APPEND EMB_BASELINE "\t{ \"${key}\", ${ms} },\n")
endforeach()
string(APPEND EMB_BASELINE "};\n")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/emb_baseline.h "${EMB_BASELINE}")

set(MAIN_SRCS okapiabench.c bench.c recap.c)
set(INT_SRCS cpu.c toolbox.c quickdraw.c cquickdraw.c apps.c imgenc_soft.c)
set(FPU_SRCS fp.c imgenc_fpu.c)
if(NOT PPC)
    list(APPEND FPU_SRCS fpu68881.c)
endif()
set(MINIZ_SRCS ${MINIZ}/miniz.c ${MINIZ}/miniz_tdef.c ${MINIZ}/miniz_tinfl.c ${MINIZ}/miniz_zip.c)
set(COREMARK_SRCS ${COREMARK}/core_list_join.c ${COREMARK}/core_main.c ${COREMARK}/core_matrix.c
    ${COREMARK}/core_state.c ${COREMARK}/core_util.c coremark_port/core_portme.c)

set(EMB_SRCS)
if(NOT NO_EMBENCH)
    file(GLOB EMB_DIRS LIST_DIRECTORIES true ${EMBENCH}/src/*)
    foreach(dir ${EMB_DIRS})
        if(NOT IS_DIRECTORY ${dir})
            continue()
        endif()
        get_filename_component(bench ${dir} NAME)
        if(bench STREQUAL "dummy-benchmark")
            continue()
        endif()
        string(REPLACE "-" "_" id ${bench})
        file(GLOB srcs ${dir}/*.c)
        # One application, nineteen programs: each one's four entry points get its name.
        set_source_files_properties(${srcs} PROPERTIES
            COMPILE_OPTIONS "${SOFT_FLAGS};-std=gnu11;-w;-include;${CMAKE_CURRENT_SOURCE_DIR}/emb_config.h"
            COMPILE_DEFINITIONS "benchmark=emb_${id}_benchmark;initialise_benchmark=emb_${id}_initialise_benchmark;warm_caches=emb_${id}_warm_caches;verify_benchmark=emb_${id}_verify_benchmark"
            INCLUDE_DIRECTORIES "${EMBENCH}/support;${dir}")
        list(APPEND EMB_SRCS ${srcs})
    endforeach()
    set_source_files_properties(${EMBENCH}/support/beebsc.c PROPERTIES
        COMPILE_OPTIONS "${SOFT_FLAGS};-std=gnu11;-w" INCLUDE_DIRECTORIES "${EMBENCH}/support")
    list(APPEND EMB_SRCS ${EMBENCH}/support/beebsc.c emb_port.c)
endif()

add_application(OkapiaBench ${MAIN_SRCS} ${INT_SRCS} ${FPU_SRCS} ${MINIZ_SRCS} ${COREMARK_SRCS} ${EMB_SRCS}
    okapiabench.r CREATOR OKBN)

target_include_directories(OkapiaBench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(OkapiaBench PRIVATE -O2 -ffunction-sections)
if(NO_EMBENCH)
    target_compile_definitions(OkapiaBench PRIVATE NO_EMBENCH)
endif()
if(REFERENCE)
    configure_file(${REFERENCE} ${CMAKE_CURRENT_BINARY_DIR}/reference.h COPYONLY)
    target_compile_definitions(OkapiaBench PRIVATE OKB_REFERENCE)
endif()
if(PPC)
    target_compile_options(OkapiaBench PRIVATE -ffp-contract=off)
endif()

set_source_files_properties(${INT_SRCS} emb_port.c PROPERTIES COMPILE_OPTIONS "${INT_FLAGS}")
set_source_files_properties(${FPU_SRCS} PROPERTIES COMPILE_OPTIONS "${FPU_FLAGS}")
set_source_files_properties(apps.c imgenc_soft.c imgenc_fpu.c PROPERTIES INCLUDE_DIRECTORIES "${STB};${MINIZ};${CMAKE_CURRENT_SOURCE_DIR}/shims")
set_source_files_properties(apps.c PROPERTIES COMPILE_OPTIONS "${INT_FLAGS};-Wno-unused-function")
set_source_files_properties(${MINIZ_SRCS} PROPERTIES
    COMPILE_OPTIONS "${INT_FLAGS};-w" COMPILE_DEFINITIONS "MINIZ_NO_STDIO;MINIZ_NO_TIME"
    INCLUDE_DIRECTORIES "${MINIZ};${CMAKE_CURRENT_SOURCE_DIR}/shims")
set_source_files_properties(apps.c PROPERTIES COMPILE_DEFINITIONS "MINIZ_NO_STDIO;MINIZ_NO_TIME")
if(PPC)
    set(COREMARK_FLAGS_STR "-O2 -ffp-contract=off")
else()
    set(COREMARK_FLAGS_STR "-O2 -m68020 -msoft-float")
endif()
set_source_files_properties(${COREMARK_SRCS} PROPERTIES
    COMPILE_OPTIONS "${INT_FLAGS}"
    COMPILE_DEFINITIONS "FLAGS_STR=\"${COREMARK_FLAGS_STR}\""
    INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/coremark_port;${COREMARK}")
# CoreMark's main is one of OkapiaBench's tests (coremark_port/core_portme.c)
set_property(SOURCE ${COREMARK}/core_main.c APPEND PROPERTY COMPILE_DEFINITIONS "main=coremark_main")

target_link_libraries(OkapiaBench m)
if(PPC)
    set_target_properties(OkapiaBench PROPERTIES LINK_FLAGS "-Wl,-gc-sections")
else()
    set_target_properties(OkapiaBench PROPERTIES LINK_FLAGS "-Wl,-gc-sections -Wl,--mac-single")
endif()
